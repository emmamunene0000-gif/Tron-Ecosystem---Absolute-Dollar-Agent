get_line_type(_style) =>
    switch _style
        "Solid" => line.style_solid
        "Dotted" => line.style_dotted
        "Dashed" => line.style_dashed

get_size(x) =>
    switch x
        "Auto" => size.auto
        "Tiny" => size.tiny
        "Small" => size.small
        "Normal" => size.normal
        "Large" => size.large
        "Huge" => size.huge

get_table_pos(pos) =>
    switch pos
        "Bottom Center" => position.bottom_center
        "Bottom Left" => position.bottom_left
        "Bottom Right" => position.bottom_right
        "Middle Center" => position.middle_center
        "Middle Left" => position.middle_left
        "Middle Right" => position.middle_right
        "Top Center" => position.top_center
        "Top Left" => position.top_left
        "Top Right" => position.top_right

leg(int size) =>
    var leg     = 0
    newLegHigh  = high[size] > ta.highest(size)
    newLegLow   = low[size]  < ta.lowest(size)
    if newLegHigh
        leg := BEARISH_LEG
    else if newLegLow
        leg := BULLISH_LEG
    leg

startOfNewLeg(int leg)     => ta.change(leg) != 0
startOfBearishLeg(int leg) => ta.change(leg) == -1
startOfBullishLeg(int leg) => ta.change(leg) == +1

getCurrentStructure(int size, bool equalHighLow=false, bool internal=false) =>
    currentLeg  = leg(size)
    newPivot    = startOfNewLeg(currentLeg)
    pivotLow    = startOfBullishLeg(currentLeg)
    pivotHigh   = startOfBearishLeg(currentLeg)
    if newPivot
        if pivotLow
            pivot p_ivot = equalHighLow ? equalLow : internal ? internalLow : swingLow
            if equalHighLow and math.abs(p_ivot.currentLevel - low[size]) < equalHighsLowsThresholdInput * atrMeasure
                drawEqualHighLow(p_ivot, low[size], size, false)
                currentAlerts.equalLows := true
            p_ivot.lastLevel    := p_ivot.currentLevel
            p_ivot.currentLevel := low[size]
            p_ivot.crossed      := false
            p_ivot.barTime      := time[size]
            p_ivot.barIndex     := bar_index[size]
            if not equalHighLow and not internal
                trailing.bottom         := p_ivot.currentLevel
                trailing.barTime        := p_ivot.barTime
                trailing.barIndex       := p_ivot.barIndex
                trailing.lastBottomTime := p_ivot.barTime
            if showSwingsInput and not internal and not equalHighLow
                drawLabel(time[size], p_ivot.currentLevel, p_ivot.currentLevel < p_ivot.lastLevel ? 'LL' : 'HL', swingBullishColor, label.style_label_up)
        else
            pivot p_ivot = equalHighLow ? equalHigh : internal ? internalHigh : swingHigh
            if equalHighLow and math.abs(p_ivot.currentLevel - high[size]) < equalHighsLowsThresholdInput * atrMeasure
                drawEqualHighLow(p_ivot, high[size], size, true)
                currentAlerts.equalHighs := true
            p_ivot.lastLevel    := p_ivot.currentLevel
            p_ivot.currentLevel := high[size]
            p_ivot.crossed      := false
            p_ivot.barTime      := time[size]
            p_ivot.barIndex     := bar_index[size]
            if not equalHighLow and not internal
                trailing.top         := p_ivot.currentLevel
                trailing.barTime     := p_ivot.barTime
                trailing.barIndex    := p_ivot.barIndex
                trailing.lastTopTime := p_ivot.barTime
            if showSwingsInput and not internal and not equalHighLow
                drawLabel(time[size], p_ivot.currentLevel, p_ivot.currentLevel > p_ivot.lastLevel ? 'HH' : 'LH', swingBearishColor, label.style_label_down)

deleteOrderBlocks(bool internal=false) =>
    array<orderBlock> orderBlocks = internal ? internalOrderBlocks : swingOrderBlocks
    for [index,eachOB] in orderBlocks
        bool crossed = false
        if bearishOrderBlockMitigationSource > eachOB.barHigh and eachOB.bias == BEARISH
            crossed := true
            if internal
                currentAlerts.internalBearishOrderBlock := true
            else
                currentAlerts.swingBearishOrderBlock    := true
        else if bullishOrderBlockMitigationSource < eachOB.barLow and eachOB.bias == BULLISH
            crossed := true
            if internal
                currentAlerts.internalBullishOrderBlock := true
            else
                currentAlerts.swingBullishOrderBlock    := true
        if crossed
            orderBlocks.remove(index)

storeOrderBlock(pivot p_ivot, bool internal=false, int bias=1) =>
    if (not internal and showSwingOrderBlocksInput) or (internal and showInternalOrderBlocksInput)
        array<float> a_rray = na
        int parsedIndex = na
        if bias == BEARISH
            a_rray      := parsedHighs.slice(p_ivot.barIndex, bar_index)
            parsedIndex := p_ivot.barIndex + a_rray.indexof(a_rray.max())
        else
            a_rray      := parsedLows.slice(p_ivot.barIndex, bar_index)
            parsedIndex := p_ivot.barIndex + a_rray.indexof(a_rray.min())
        orderBlock ob             = orderBlock.new(parsedHighs.get(parsedIndex), parsedLows.get(parsedIndex), times_arr.get(parsedIndex), bias)
        array<orderBlock> blocks  = internal ? internalOrderBlocks : swingOrderBlocks
        if blocks.size() >= 100
            blocks.pop()
        blocks.unshift(ob)

drawFairValueGaps() =>
    [lastClose, lastOpen, lastTime, currentHigh, currentLow, currentTime, last2High, last2Low] = request.security(syminfo.tickerid, fairValueGapsTimeframeInput, [close[1],open[1],time[1],high[0],low[0],time[0],high[2],low[2]], lookahead=barmerge.lookahead_off)
    barDeltaPercent     = (lastClose - lastOpen) / (lastOpen * 100)
    newTimeframe        = timeframe.change(fairValueGapsTimeframeInput)
    threshold           = fairValueGapsThresholdInput ? ta.cum(math.abs(newTimeframe ? barDeltaPercent : 0)) / bar_index * 2 : 0
    bullishFVG          = currentLow > last2High and lastClose > last2High and barDeltaPercent > threshold and newTimeframe

        currentAlerts.bullishFairValueGap := true
        fairValueGaps.unshift(fairValueGap.new(currentLow, last2High, BULLISH, fairValueGapBox(lastTime, currentTime, currentLow, math.avg(currentLow,last2High), fairValueGapBullishColor), fairValueGapBox(lastTime, currentTime, math.avg(currentLow,last2High), last2High, fairValueGapBullishColor)))
    if bearishFVG
        currentAlerts.bearishFairValueGap := true
        fairValueGaps.unshift(fairValueGap.new(currentHigh, last2Low, BEARISH, fairValueGapBox(lastTime, currentTime, currentHigh, math.avg(currentHigh,last2Low), fairValueGapBearishColor), fairValueGapBox(lastTime, currentTime, math.avg(currentHigh,last2Low), last2Low, fairValueGapBearishColor)))

lq_get_counts(condition, top, btm) =>
    var count = 0

        count := 0
        vol   := 0.
    else
        if lq_intraPrecision
            if bar_index > lq_length
                if array.size(lq_v[lq_length]) > 0
                    for [index, element] in lq_v[lq_length]
                        vol += array.get(lq_l[lq_length], index) < top and array.get(lq_h[lq_length], index) > btm ? element : 0
        else
            vol += low[lq_length] < top and high[lq_length] > btm ? volume[lq_length] : 0
        count += low[lq_length] < top and high[lq_length] > btm ? 1 : 0
    [count, vol]

        'Count'  => count
        'Volume' => vol
    if ta.crossover(target, lq_filterValue)
        lbl := label.new(x, y, str.tostring(vol, format.volume), style=lbl_style, size=label_size, color=#00000000, textcolor=css)

update_dwm_info(dwm_info info) =>
    if timeframe.change(info.tf)
        info.ph := info.h
        info.pl := info.l
        info.o := open
        info.h := high
        info.l := low
    else
        info.h := math.max(high, info.h)
        info.l := math.min(low, info.l)

dwm_sep(string tf, bool use, array<line> arr, color col) =>
    if use
        if timeframe.change(tf)
            arr.unshift(line.new(bar_index, high * 1.0001, bar_index, low, style = htf_style, width = htf_width, extend = extend.both, color = col))
            if not sep_unlimited and arr.size() > max_days
                arr.pop().delete()

dwm_open(string tf, bool use, array<line> lns, array<label> lbls, dwm_info info, color col) =>
    if use
        if lns.size() > 0
            lns.get(0).set_x2(time)
            lbls.get(0).set_x(time)
        if timeframe.change(tf)
            lns.unshift(line.new(time, info.o, time, info.o, xloc = xloc.bar_time, style = htf_style, width = htf_width, color = col))
            lbls.unshift(label.new(time, info.o, tf + " OPEN", xloc = xloc.bar_time, style = label.style_label_left, color = transparent, textcolor = txt_color, size = kz_lbl_size))
            if not sep_unlimited and lns.size() > max_days
                lns.pop().delete()
                lbls.pop().delete()

dwm_hl(string tf, bool use, dwm_hl hl, dwm_info info, color col) =>
    if use
        if hl.hi_line.size() > 0
            hl.hi_line.get(0).set_x2(time)
            hl.lo_line.get(0).set_x2(time)
            hl.hi_label.get(0).set_x(time)
            hl.lo_label.get(0).set_x(time)
        if timeframe.change(tf)
            hl.hi_line.unshift(line.new(time, info.ph, time, info.ph, xloc = xloc.bar_time, style = htf_style, width = htf_width, color = col))
            hl.lo_line.unshift(line.new(time, info.pl, time, info.pl, xloc = xloc.bar_time, style = htf_style, width = htf_width, color = col))
            hl.hi_label.unshift(label.new(time, info.ph, "P" + tf + "H", xloc = xloc.bar_time, style = label.style_label_left, color = transparent, textcolor = txt_color, size = kz_lbl_size))
            hl.lo_label.unshift(label.new(time, info.pl, "P" + tf + "L", xloc = xloc.bar_time, style = label.style_label_left, color = transparent, textcolor = txt_color, size = kz_lbl_size))
            hl.hit_high := false
            hl.hit_low := false
            if not sep_unlimited and hl.hi_line.size() > max_days
                hl.hi_line.pop().delete()
                hl.lo_line.pop().delete()
                hl.hi_label.pop().delete()
                hl.lo_label.pop().delete()
        if hl.hi_line.size() > 0 and alert_HL
            if not hl.hit_high and high > hl.hi_line.get(0).get_y1()
                hl.hit_high := true
                alert(str.format("Hit P{0}H", tf))
            if not hl.hit_low and low < hl.lo_line.get(0).get_y1()
                hl.hit_low := true
                alert(str.format("Hit P{0}L", tf))

method hz_line(lines_helper this) =>
    bool t = not na(time("", this.h, gmt_tz))
    bool t_prev = not na(time("", this.h, gmt_tz, bars_back = 1))
    hz hzObj = this._hz

        hzObj.LB.unshift(label.new(bar_index, open, txt, style = label.style_label_left, color = transparent, textcolor = txt_color, size = kz_lbl_size))

            hzObj.LB.pop().delete()
            hzObj.CO.pop()

            hzObj.LN.get(0).set_x2(bar_index)
            hzObj.LB.get(0).set_x(bar_index)

del_kz(kz kObj) =>
    if kObj._box.size() > max_days
        kObj._box.pop().delete()
    if kObj._hi_line.size() > max_days
        kObj._hi_line.pop().delete()

        kObj._hi_valid.pop()
        kObj._lo_valid.pop()
        if show_midpoints
            kObj._md_line.pop().delete()

        kObj._lo_label.pop().delete()

update_price_string(label L, float P) =>
    S = L.get_text()
    pre = str.substring(S, 0, str.pos(S, " ")) 
    str.trim(pre)
    L.set_text(str.format("{0} ({1})", pre, P))

adjust_in_kz(kz kObj, bool t) =>
    if t
        kzBox0 = kObj._box.get(0)
        kzBox0.set_right(time)
        newTop = math.max(kzBox0.get_top(), high)
        kzBox0.set_top(newTop)
        newBottom = math.min(kzBox0.get_bottom(), low)
        kzBox0.set_bottom(newBottom)
        kObj._range_current := newTop - newBottom
        if show_pivots and kObj._hi_line.size() > 0
            kzHiLine0 = kObj._hi_line.get(0)
            kzHiLine0.set_x2(time)
            if high > kzHiLine0.get_y1()
                kzHiLine0.set_xy1(time, high)
                kzHiLine0.set_xy2(time, high)
            kzLoLine0 = kObj._lo_line.get(0)
            kzLoLine0.set_x2(time)
            if low < kzLoLine0.get_y1()
                kzLoLine0.set_xy1(time, low)
                kzLoLine0.set_xy2(time, low)
            if show_midpoints
                kzMidLine0 = kObj._md_line.get(0)
                kzMidLine0.set_x2(time)
                kzMidLine0.set_xy1(time, math.avg(kzHiLine0.get_y2(), kzLoLine0.get_y2()))
                kzMidLine0.set_xy2(time, math.avg(kzHiLine0.get_y2(), kzLoLine0.get_y2()))
        if show_labels and kObj._hi_label.size() > 0
            if label_right
                kObj._hi_label.get(0).set_x(time)
                kObj._lo_label.get(0).set_x(time)
            if high > kObj._hi_label.get(0).get_y()
                kObj._hi_label.get(0).set_xy(time, high)
                if label_price
                    update_price_string(kObj._hi_label.get(0), high)
            if low < kObj._lo_label.get(0).get_y()
                kObj._lo_label.get(0).set_xy(time, low)
                if label_price
                    update_price_string(kObj._lo_label.get(0), low)

adjust_out_kz(kz kObj, bool t, bool t_prev) =>
    boxCount = kObj._box.size()
    if not t and boxCount > 0
        if t_prev
            array.unshift(kObj._range_store, kObj._range_current)
            if kObj._range_store.size() > range_avg
                kObj._range_store.pop()
    if show_pivots and boxCount > 0
        for i = 0 to boxCount - 1 by 1
            if not ext_current or i == 0
                kzHiValid = kObj._hi_valid.get(i)
                if ext_past or kzHiValid
                    kObj._hi_line.get(i).set_x2(time)
                    if show_labels and label_right
                        kObj._hi_label.get(i).set_x(time)
                if kzHiValid and high > kObj._hi_line.get(i).get_y1()
                    if use_alerts and i == 0
                        alert('Broke ' + kObj._title + ' High', alert.freq_once_per_bar)
                    kObj._hi_valid.set(i, false)
                    if show_labels and label_right
                        kObj._hi_label.get(i).set_style(label.style_label_down)
                else if use_cutoff ? t_co : false
                    kObj._hi_valid.set(i, false)
                kzLoValid = kObj._lo_valid.get(i)
                if ext_past or kzLoValid
                    kObj._lo_line.get(i).set_x2(time)
                    if show_labels and label_right
                        kObj._lo_label.get(i).set_x(time)
                if kzLoValid and low < kObj._lo_line.get(i).get_y1()
                    if use_alerts and i == 0
                        alert("Broke " + kObj._title + " Low", alert.freq_once_per_bar)
                    kObj._lo_valid.set(i, false)
                    if show_labels and label_right
                        kObj._lo_label.get(i).set_style(label.style_label_up)
                else if use_cutoff ? t_co : false
                    kObj._lo_valid.set(i, false)
                if show_midpoints and not t
                    if stop_midpoints ? kObj._md_valid.get(i) : true
                        kObj._md_line.get(i).set_x2(time)
                        if kObj._md_valid.get(i) and low <= kObj._md_line.get(i).get_y1() and high >= kObj._md_line.get(i).get_y1()
                            kObj._md_valid.set(i, false)
            else
                break

method manage_kz(kz_helper this) =>
    kz kObj = this._kz    
    c = this.c
    string box_txt = this.box_txt
    string hi_txt = this.hi_txt
    string loTxt = this.lo_txt
    if tf_limit_is_equal_or_more_chart_tf
        t = not na(time("", this.session, gmt_tz))
        t_prev = not na(time("", this.session, gmt_tz, bars_back = 1))
        if t and not t_prev
            _c = get_box_color(c)
            _t = get_text_color(c)
            kObj._box.unshift(box.new(time, high, time, low, xloc = xloc.bar_time, border_color = show_kz ? _c : na, bgcolor = show_kz ? _c : na, text = show_kz and show_kz_text ? box_txt : na, text_color = _t))
            if show_pivots
                kObj._hi_line.unshift(line.new(time, high, time, high, xloc = xloc.bar_time, style = kzp_style, color = c, width = kzp_width))
                kObj._lo_line.unshift(line.new(time, low, time, low, xloc = xloc.bar_time, style = kzp_style, color = c, width = kzp_width))
                if show_midpoints
                    kObj._md_line.unshift(line.new(time, math.avg(high, low), time, math.avg(high, low), xloc = xloc.bar_time, style = kzm_style, color = c, width = kzm_width))
                    array.unshift(kObj._md_valid, true)
                array.unshift(kObj._hi_valid, true)
                array.unshift(kObj._lo_valid, true)
                if show_labels
                    _hi_txt = label_price ? str.format('{0} ({1})', hi_txt, high) : hi_txt
                    _lo_txt = label_price ? str.format('{0} ({1})', loTxt, low) : loTxt
                    if label_right
                        kObj._hi_label.unshift(label.new(time, high, _hi_txt, xloc = xloc.bar_time, color = transparent, textcolor = txt_color, style = label.style_label_left, size = kz_lbl_size))
                        kObj._lo_label.unshift(label.new(time, low, _lo_txt, xloc = xloc.bar_time, color = transparent, textcolor = txt_color, style = label.style_label_left, size = kz_lbl_size))
                    else
                        kObj._hi_label.unshift(label.new(time, high, _hi_txt, xloc = xloc.bar_time, color = transparent, textcolor = txt_color, style = label.style_label_down, size = kz_lbl_size))
                        kObj._lo_label.unshift(label.new(time, low, _lo_txt, xloc = xloc.bar_time, color = transparent, textcolor = txt_color, style = label.style_label_up, size = kz_lbl_size))
            del_kz(kObj)
        adjust_in_kz(kObj, t)
        adjust_out_kz(kObj, t, t_prev)

get_min_days_stored() =>
    store = array.new_int()
    for [_, value] in _kz
        int tmpStoreSize = value._kz._range_store.size()
        if tmpStoreSize > 0
            store.push(tmpStoreSize) 
    store.min()

method set_table(table tbl, kz kObj, int row, string txt, bool t, color col) =>

    table.cell(tbl, 1, row, str.tostring(kObj._range_current), text_size = range_size, bgcolor = t ? get_box_color(col) : na, text_color = txt_color)
    if show_range_avg
        table.cell(tbl, 2, row, str.tostring(kObj._range_store.avg()), text_size = range_size, text_color = txt_color)
